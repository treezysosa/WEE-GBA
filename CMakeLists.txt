cmake_minimum_required(VERSION 3.22)
project(WEE VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(WEE_CORE_ONLY "Build portable DSP tests without JUCE" OFF)
include(CTest)
if(BUILD_TESTING)
  add_executable(wee_dsp_tests tests/DspTests.cpp)
  target_include_directories(wee_dsp_tests PRIVATE Source)
  add_test(NAME wee_dsp COMMAND wee_dsp_tests)
endif()
if(WEE_CORE_ONLY)
  return()
endif()
set(WEE_JUCE_PATH "" CACHE PATH "Optional existing JUCE 8.0.6 checkout")
if(WEE_JUCE_PATH)
  add_subdirectory("${WEE_JUCE_PATH}" JUCE)
else()
  include(FetchContent)
  FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.6 GIT_SHALLOW TRUE)
  FetchContent_MakeAvailable(JUCE)
endif()
option(WEE_ENABLE_ASIO "Enable ASIO; requires a separately licensed ASIO SDK" OFF)
if(WEE_ENABLE_ASIO)
  set(WEE_ASIO_SDK_PATH "" CACHE PATH "ASIO SDK root containing common/asio.h")
  if(NOT EXISTS "${WEE_ASIO_SDK_PATH}/common/asio.h")
    message(FATAL_ERROR "Set WEE_ASIO_SDK_PATH to an installed ASIO SDK")
  endif()
endif()
juce_add_plugin(WEE COMPANY_NAME "WEE Audio" BUNDLE_ID "audio.wee.instrument"
  PLUGIN_MANUFACTURER_CODE WeeA PLUGIN_CODE Wee1 FORMATS VST3 Standalone
  PRODUCT_NAME "WEE" IS_SYNTH TRUE NEEDS_MIDI_INPUT TRUE NEEDS_MIDI_OUTPUT FALSE
  IS_MIDI_EFFECT FALSE EDITOR_WANTS_KEYBOARD_FOCUS TRUE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(WEE PRIVATE Source/PluginProcessor.cpp Source/ui/PluginEditor.cpp)
target_include_directories(WEE PUBLIC Source)
if(WEE_ENABLE_ASIO)
  target_include_directories(WEE PRIVATE "${WEE_ASIO_SDK_PATH}/common")
endif()
target_compile_definitions(WEE PUBLIC JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0
  JUCE_VST3_CAN_REPLACE_VST2=0 JUCE_ASIO=$<BOOL:${WEE_ENABLE_ASIO}>)
target_link_libraries(WEE PRIVATE juce::juce_audio_utils juce::juce_dsp
  PUBLIC juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
if(BUILD_TESTING)
  juce_add_console_app(wee_state_tests PRODUCT_NAME "WEE state tests")
  target_sources(wee_state_tests PRIVATE tests/StateTests.cpp)
  target_link_libraries(wee_state_tests PRIVATE WEE juce::juce_audio_utils juce::juce_dsp)
  add_test(NAME wee_state COMMAND wee_state_tests)
endif()
